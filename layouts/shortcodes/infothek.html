{{- /* The Infothek: everything in data/books/, in book view, as one list: books and publications (no type) first, then podcasts, then videos. {{< infothek >}} all entries {{< infothek "hoeren" >}} only one type: lesen (books), hoeren (Podcast) or schauen (Video) "weight" sets the order within a type (smaller = higher); entries without one go last. Filterable with {{< media-filter "infothek" >}} on the same page. */ -}} {{- $types := dict "lesen" "" "hoeren" "Podcast" "schauen" "Video" -}} {{- $sections := slice "lesen" "hoeren" "schauen" -}} {{- with .Get 0 -}} {{- if not (in $sections .) -}} {{- warnf "infothek: unknown section %q in %s (lesen, hoeren or schauen)" . $.Position -}} {{- end -}} {{- $sections = slice . -}} {{- end -}}
{{- range $section := $sections }} {{- $items := slice -}} {{- range hugo.Data.books -}} {{- if eq (.type | default "") (index $types $section) -}} {{- $items = $items | append (merge (dict "weight" 9999) .) -}} {{- end -}} {{- end -}} {{- range sort $items "weight" }} {{ partial "media-item" (dict "item" . "section" $section) }} {{- else -}} {{- warnf "infothek: no entries for %q in %s" $section $.Position -}} {{- end }} {{- end }}
