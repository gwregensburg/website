{{/* Press articles from data/press/, edited in the CMS. {{< press topic="plk" >}} all articles with this topic, newest first {{< press "" >}} a single article (id = file name without .yaml) */}} {{- with .Get "topic" -}} {{- $topic := . -}} {{- $found := slice -}} {{- range $id, $article := hugo.Data.press -}} {{- if in ($article.topics | default slice) $topic -}} {{- $found = $found | append (dict "id" $id "date" $article.date "article" $article) -}} {{- end -}} {{- end -}} {{- range sort $found "date" "desc" -}} {{ partial "press-article" . }} {{- else -}} {{- warnf "press: no articles with topic %q in %s" $topic $.Position -}} {{- end -}} {{- else -}} {{- $id := .Get 0 | default (.Get "id") -}} {{- with index hugo.Data.press $id -}} {{ partial "press-article" (dict "id" $id "article" .) }} {{- else -}} {{- warnf "press: unknown article %q in %s" $id $.Position -}} {{- end -}} {{- end -}}
